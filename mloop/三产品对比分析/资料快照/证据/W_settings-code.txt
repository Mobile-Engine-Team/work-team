00098  /** Map a legacy settings tab id (first path segment) to a hi-fi section. */
00099  export function resolveMwSettingsSectionFromLegacyTab(tab: string): MwSettingsSectionKey {
00100    const key = tab.split("/")[0]?.trim() ?? "";
00101    if (isMwSettingsSectionKey(key)) return key;
00102    return LEGACY_SETTINGS_TAB_TO_MW[key] ?? "account";
