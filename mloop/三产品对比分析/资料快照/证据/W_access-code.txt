00755  function assertOpencodeProxyAllowed(actor: Actor, method: string, proxyPath: string) {
00756    const m = method.toUpperCase();
00757    const scope = actor.scope ?? "viewer";
00758  
00759    // 百分号编码与重复斜杠都不能绕过生命周期端点拦截。
00760    let decodedPath = proxyPath;
00761    try {
00762      decodedPath = decodeURIComponent(proxyPath);
00763    } catch {
00764      // 非法编码按原文匹配，交给上游拒绝。
00765    }
00766    const lifecyclePath = normalizeOpencodeProxyPath(decodedPath.replace(/\/{2,}/g, "/"));
00767    if (ENGINE_LIFECYCLE_PROXY_ENDPOINTS.some(([endpointMethod, path]) => endpointMethod === m && path.test(lifecyclePath))) {
00768      throw new ApiError(403, "engine_lifecycle_proxy_forbidden", "引擎实例释放与配置写入须经服务端统一重载");
00769    }
00770  
00771    if (scope === "viewer" && m !== "GET" && m !== "HEAD") {
00772      throw new ApiError(403, "forbidden", "Viewer tokens are read-only");
00773    }
00774  
00775    // 审批、沙箱策略和会话权限模式只能由所有者修改；共享同一套路径解码护栏。
00776    if (scope !== "owner" && m !== "GET" && m !== "HEAD") {
00777      const normalized = lifecyclePath;
00778      if (/\/permission\/[^/]+\/reply$/.test(normalized)
00779        || /^\/sandbox(?:\/|$)/.test(normalized)
00780        || /^\/session\/[^/]+\/permission-mode$/.test(normalized)) {
00781        throw new ApiError(403, "forbidden", "仅所有者可以审批或修改沙箱与会话权限");
00782      }
00783    }
00784  }
